local flying = false

local FLY_SPEED_MIN = 0
local FLY_SPEED_MAX = 100
local FLY_SPEED_STEP = 10

local renderConnection = nil

-- ===== 飞行逻辑 =====

local function startFly()
	print("[FlightController] 尝试开始飞行, 角色状态:", player.Character ~= nil)
	local character = player.Character
	if not character then
		warn("[FlightController] 没有角色, 飞行失败")
		return
	end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local rootPart = character:FindFirstChild("HumanoidRootPart")
	if not humanoid or not rootPart then
		warn("[FlightController] 缺少 Humanoid 或 HumanoidRootPart")
		return
	end
	print("[FlightController] 飞行启动中...")

	-- 清理旧的
	if bodyVelocity then bodyVelocity:Destroy() end
	if bodyGyro then bodyGyro:Destroy() end

	humanoid.PlatformStand = true

	bodyVelocity = Instance.new("BodyVelocity")
	bodyVelocity.MaxForce = Vector3.new(1, 1, 1) * 1e5
	bodyVelocity.Velocity = Vector3.zero
	bodyVelocity.Parent = rootPart

	bodyGyro = Instance.new("BodyGyro")
	bodyGyro.MaxTorque = Vector3.new(1, 1, 1) * 1e5
	bodyGyro.P = 1e4
	bodyGyro.CFrame = rootPart.CFrame
	bodyGyro.Parent = rootPart

	flying = true
	updateSpeedDisplay()
	local walkSpeed = humanoid.WalkSpeed > 0 and humanoid.WalkSpeed or 16
	print("[FlightController] ✓ 飞行中! 倍率:", flySpeed, "实际速度:", flySpeed * walkSpeed, "studs/s")

	-- 每帧更新飞行方向和速度
	if renderConnection then renderConnection:Disconnect() end
	renderConnection = RunService.RenderStepped:Connect(function()
		if not flying or not bodyVelocity or not bodyGyro then return end

		local char = player.Character
		if not char then return end
		local root = char:FindFirstChild("HumanoidRootPart")
		if not root then return end

		local cam = workspace.CurrentCamera
		if not cam then return end

		local camCFrame = cam.CFrame
		local forward = camCFrame.LookVector
		local right = camCFrame.RightVector
		local up = Vector3.new(0, 1, 0)

		local moveVec = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec += forward end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec -= forward end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec -= right end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec += right end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVec += up end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveVec -= up end

		if moveVec.Magnitude > 0 then
			-- flySpeed 是倍率: 1 = 走路速度(WalkSpeed), 数值越大越快
			local walkSpeed = humanoid.WalkSpeed > 0 and humanoid.WalkSpeed or 16
			moveVec = moveVec.Unit * (flySpeed * walkSpeed)
		end

		bodyVelocity.Velocity = moveVec
		bodyGyro.CFrame = camCFrame
	end)
end

local function stopFly()
	print("[FlightController] 停止飞行")
	local character = player.Character
	if character then
		local humanoid = character:FindFirstChildOfClass("Humanoid")
		if humanoid then
			humanoid.PlatformStand = false
		end
	end

	if bodyVelocity then
		bodyVelocity:Destroy()
		bodyVelocity = nil
	end
	if bodyGyro then
		bodyGyro:Destroy()
		bodyGyro = nil
	end
	if renderConnection then
		renderConnection:Disconnect()
		renderConnection = nil
	end

	flying = false
	updateSpeedDisplay()  -- 降落后显示速度为 0
end

local function toggleFly()
	if flying then
		stopFly()
	else
		startFly()
	end
end

	flySpeed = math.clamp(math.floor(newSpeed), FLY_SPEED_MIN, FLY_SPEED_MAX)

-- 平滑切换展开/折叠
local function tweenPanel(toCollapsed)
	if not panel or not flyBtn or not stopBtn or not hintLabel or not dividerLine or not minBtn then
		return
	end
	collapsed = toCollapsed
	local targetSize = collapsed and panelMiniSize or panelFullSize

	-- 标题和按钮的淡入淡出
	local tweenInfo = TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
	TweenService:Create(panel, tweenInfo, { Size = targetSize }):Play()

	-- 需要淡入淡出的额外元素
	local extraElems = { dividerLine2, speedRow, speedLabel }
	if collapsed then
		-- 收起: 按钮/分隔线/提示淡出, 标题缩到一行
		TweenService:Create(dividerLine, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
		TweenService:Create(flyBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play()
		TweenService:Create(stopBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play()
		TweenService:Create(hintLabel, TweenInfo.new(0.2), { TextTransparency = 1 }):Play()
		for _, el in ipairs(extraElems) do
			if el then
				local targetT = 1
				if el:IsA("TextLabel") or el:IsA("TextBox") then
					TweenService:Create(el, TweenInfo.new(0.2), { TextTransparency = targetT }):Play()
				end
				if el.BackgroundTransparency < 1 then
					TweenService:Create(el, TweenInfo.new(0.2), { BackgroundTransparency = targetT }):Play()
				end
			end
		end
		-- 按钮/输入框也淡出
		if speedDownBtn then TweenService:Create(speedDownBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
		if speedUpBtn then TweenService:Create(speedUpBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
		if speedTextBox then TweenService:Create(speedTextBox, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
		task.wait(0.2)
		dividerLine.Visible = false
		flyBtn.Visible = false
		stopBtn.Visible = false
		hintLabel.Visible = false
		for _, el in ipairs(extraElems) do if el then el.Visible = false end end
		if speedDownBtn then speedDownBtn.Visible = false end
		if speedUpBtn then speedUpBtn.Visible = false end
		if speedTextBox then speedTextBox.Visible = false end
		minBtn.Text = "+"
	else
		-- 展开: 标题回到完整, 按钮淡入
		dividerLine.Visible = true
		flyBtn.Visible = true
		stopBtn.Visible = true
		hintLabel.Visible = true
		for _, el in ipairs(extraElems) do if el then el.Visible = true end end
		if speedDownBtn then speedDownBtn.Visible = true end
		if speedUpBtn then speedUpBtn.Visible = true end
		if speedTextBox then speedTextBox.Visible = true end
		TweenService:Create(dividerLine, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()
		TweenService:Create(flyBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play()
		TweenService:Create(stopBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play()
		TweenService:Create(hintLabel, TweenInfo.new(0.3), { TextTransparency = 0 }):Play()
		for _, el in ipairs(extraElems) do
			if el then
				if el:IsA("TextLabel") or el:IsA("TextBox") then
					TweenService:Create(el, TweenInfo.new(0.3), { TextTransparency = 0 }):Play()
				end
				if el.BackgroundTransparency < 1 then
					TweenService:Create(el, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()
				end
			end
		end
		if speedDownBtn then TweenService:Create(speedDownBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
		if speedUpBtn then TweenService:Create(speedUpBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
		if speedTextBox then TweenService:Create(speedTextBox, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
		minBtn.Text = "—"
	end
end

local function updateButtonVisuals()

	if flying then
		flyBtn.BackgroundColor3 = Color3.fromRGB(80, 200, 100)
		stopBtn.BackgroundColor3 = Color3.fromRGB(120, 60, 65)
	else
		flyBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 80)
		stopBtn.BackgroundColor3 = Color3.fromRGB(160, 70, 75)
	end
end

-- 键盘快捷键 F (飞行) / H (收起面板) / R (减速) / T (加速)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.F then
		toggleFly()
	elseif input.KeyCode == Enum.KeyCode.H then
		if panel and minBtn then
			tweenPanel(not collapsed)
		end
	elseif input.KeyCode == Enum.KeyCode.R then
		changeSpeed(flySpeed - FLY_SPEED_STEP)
	elseif input.KeyCode == Enum.KeyCode.T then
		changeSpeed(flySpeed + FLY_SPEED_STEP)
	end
end)

-- 角色重生时自动停止飞行
player.CharacterAdded:Connect(function(newChar)
	print("[FlightController] 角色已重生, 重置飞行状态")
	if flying then
		stopFly()
	end
end)

print("[FlightController] ✓✓✓ 飞行系统完全加载! 按 F 或点击按钮切换飞行")
