-- ===== 飞行逻辑 (V3 机制) =====

startFly = function()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    -- 禁用 Humanoid 状态 (V3 核心: 防止物理干扰)
    for _, state in ipairs(humanoidStates) do
        hum:SetStateEnabled(state, false)
    end
    hum:ChangeState(Enum.HumanoidStateType.Swimming)

    -- 禁用动画
    local animate = char:FindFirstChild("Animate")
    if animate then animate.Disabled = true end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        track:AdjustSpeed(0)
    end

    hum.PlatformStand = true

    -- BodyGyro + BodyVelocity (V3: 保持浮空 + 朝向)
    local isR6 = hum.RigType == Enum.HumanoidRigType.R6
    local bodyPart = isR6 and char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
    if not bodyPart then bodyPart = root end

    if bodyGyro then bodyGyro:Destroy() end
    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.P = 9e4
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.CFrame = bodyPart.CFrame
    bodyGyro.Parent = bodyPart

    if bodyVelocity then bodyVelocity:Destroy() end
    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Velocity = Vector3.new(0, 0.1, 0)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Parent = bodyPart

    -- TranslateBy 移动循环 (V3 核心: 全游戏通用移动)
    tpwalking = true
    if hbConnection then hbConnection:Disconnect() end
    hbConnection = RunService.Heartbeat:Connect(function()
        if not tpwalking then return end
        local c = player.Character
        if not c then return end
        local h = c:FindFirstChildOfClass("Humanoid")
        if not h then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        if not r then return end

        -- 水平移动: TranslateBy + MoveDirection (兼容PC键盘和手机摇杆)
        if h.MoveDirection.Magnitude > 0 then
            c:TranslateBy(h.MoveDirection * flySpeed)
        end

        -- 垂直移动: CFrame (上升/下降)
        local upDown = 0
        if upPressed then upDown = upDown + 1 end
        if downPressed then upDown = upDown - 1 end
        -- PC键盘上升/下降
        if not isTouch then
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then upDown = upDown + 1 end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then upDown = upDown - 1 end
        end
        if upDown ~= 0 then
            r.CFrame = r.CFrame * CFrame.new(0, upDown * flySpeed, 0)
        end

        -- BodyGyro 跟随相机朝向
        if bodyGyro and bodyGyro.Parent then
            local cam = workspace.CurrentCamera
            if cam then bodyGyro.CFrame = cam.CFrame end
        end
    end)

    flying = true
    if upBtn then upBtn.Visible = true end
    if downBtn then downBtn.Visible = true end
    if flyBtn then flyBtn.BackgroundColor3 = Color3.fromRGB(80, 200, 100) end
    if stopBtn then stopBtn.BackgroundColor3 = Color3.fromRGB(120, 60, 65) end
    updateSpeedDisplay()
end

stopFly = function()
    local char = player.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            -- 恢复 Humanoid 状态
            for _, state in ipairs(humanoidStates) do
                hum:SetStateEnabled(state, true)
            end
            hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
            hum.PlatformStand = false
        end
        -- 恢复动画
        local animate = char:FindFirstChild("Animate")
        if animate then animate.Disabled = false end
    end

    tpwalking = false
    if hbConnection then hbConnection:Disconnect() hbConnection = nil end
    if bodyGyro then bodyGyro:Destroy() bodyGyro = nil end
    if bodyVelocity then bodyVelocity:Destroy() bodyVelocity = nil end

    flying = false
    upPressed = false
    downPressed = false
    if upBtn then upBtn.Visible = false end
    if downBtn then downBtn.Visible = false end
    if flyBtn then flyBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 80) end
    if stopBtn then stopBtn.BackgroundColor3 = Color3.fromRGB(160, 70, 75) end
    updateSpeedDisplay()
end

toggleFly = function()
    if flying then stopFly() else startFly() end
end

updateSpeedDisplay = function()
    if not speedTextBox then return end
    if flying then
        speedTextBox.Text = tostring(flySpeed)
    else
        speedTextBox.Text = "0"
    end
end

changeSpeed = function(newSpeed)
    flySpeed = math.clamp(math.floor(newSpeed), FLY_SPEED_MIN, FLY_SPEED_MAX)
    updateSpeedDisplay()
end

-- ===== UI 动画 =====

local function tweenPanel(toCollapsed)
    if not panel or not flyBtn or not stopBtn or not hintLabel or not dividerLine or not minBtn then
        return
    end
    collapsed = toCollapsed
    local targetSize = collapsed and panelMiniSize or panelFullSize

    local tweenInfo = TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    TweenService:Create(panel, tweenInfo, { Size = targetSize }):Play()

    local extraElems = { dividerLine2, speedRow, speedLabel }
    if collapsed then
        TweenService:Create(dividerLine, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
        TweenService:Create(flyBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play()
        TweenService:Create(stopBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play()
        TweenService:Create(hintLabel, TweenInfo.new(0.2), { TextTransparency = 1 }):Play()
        for _, el in ipairs(extraElems) do
            if el then
                if el:IsA("TextLabel") or el:IsA("TextBox") then
                    TweenService:Create(el, TweenInfo.new(0.2), { TextTransparency = 1 }):Play()
                end
                TweenService:Create(el, TweenInfo.new(0.2), { BackgroundTransparency = 1 }):Play()
            end
        end
        if speedDownBtn then TweenService:Create(speedDownBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
        if speedUpBtn then TweenService:Create(speedUpBtn, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
        if speedTextBox then TweenService:Create(speedTextBox, TweenInfo.new(0.2), { BackgroundTransparency = 1, TextTransparency = 1 }):Play() end
        task.wait(0.2)
        dividerLine.Visible = false
        flyBtn.Visible = false
        stopBtn.Visible = false
        hintLabel.Visible = false
        for _, el in ipairs(extraElems) do if el then el.Visible = false end end
        if speedDownBtn then speedDownBtn.Visible = false end
        if speedUpBtn then speedUpBtn.Visible = false end
        if speedTextBox then speedTextBox.Visible = false end
        minBtn.Text = "+"
    else
        dividerLine.Visible = true
        flyBtn.Visible = true
        stopBtn.Visible = true
        hintLabel.Visible = true
        for _, el in ipairs(extraElems) do if el then el.Visible = true end end
        if speedDownBtn then speedDownBtn.Visible = true end
        if speedUpBtn then speedUpBtn.Visible = true end
        if speedTextBox then speedTextBox.Visible = true end
        TweenService:Create(dividerLine, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()
        TweenService:Create(flyBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play()
        TweenService:Create(stopBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play()
        TweenService:Create(hintLabel, TweenInfo.new(0.3), { TextTransparency = 0 }):Play()
        for _, el in ipairs(extraElems) do
            if el then
                if el:IsA("TextLabel") or el:IsA("TextBox") then
                    TweenService:Create(el, TweenInfo.new(0.3), { TextTransparency = 0 }):Play()
                end
                TweenService:Create(el, TweenInfo.new(0.3), { BackgroundTransparency = 0 }):Play()
            end
        end
        if speedDownBtn then TweenService:Create(speedDownBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
        if speedUpBtn then TweenService:Create(speedUpBtn, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
        if speedTextBox then TweenService:Create(speedTextBox, TweenInfo.new(0.3), { BackgroundTransparency = 0, TextTransparency = 0 }):Play() end
        minBtn.Text = "—"
    end
end

    -- ===== 速度按钮事件 =====
    speedDownBtn.MouseButton1Click:Connect(function()
        changeSpeed(flySpeed - FLY_SPEED_STEP)
    end)
    speedUpBtn.MouseButton1Click:Connect(function()
        changeSpeed(flySpeed + FLY_SPEED_STEP)
    end)
    speedTextBox.FocusLost:Connect(function(enterPressed)
        local num = tonumber(speedTextBox.Text)
        if num then
            changeSpeed(num)
        else
            updateSpeedDisplay()
        end
    end)

    -- ===== 拖动逻辑 (AbsolutePosition, 不会消失) =====
    local dragging = false
    local dragStart = Vector2.zero
    local dragAbsStart = Vector2.zero

    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = Vector2.new(input.Position.X, input.Position.Y)
            dragAbsStart = panel.AbsolutePosition
            TweenService:Create(titleBar, TweenInfo.new(0.15), { BackgroundColor3 = Color3.fromRGB(50, 50, 70) }):Play()
        end
    end)
    titleBar.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            TweenService:Create(titleBar, TweenInfo.new(0.2), { BackgroundColor3 = Color3.fromRGB(35, 35, 50) }):Play()
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging
            and (input.UserInputType == Enum.UserInputType.MouseMovement
                or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = Vector2.new(input.Position.X - dragStart.X, input.Position.Y - dragStart.Y)
            local newAbsX = dragAbsStart.X + delta.X
            local newAbsY = dragAbsStart.Y + delta.Y
            local cam = workspace.CurrentCamera
            local view = cam and cam.ViewportSize or Vector2.new(800, 600)
            local panelSize = panel.AbsoluteSize
            local clampedX = math.clamp(newAbsX, 0, math.max(0, view.X - panelSize.X))
            local clampedY = math.clamp(newAbsY, 0, math.max(0, view.Y - panelSize.Y))
            panel.Position = UDim2.new(0, clampedX, 0, clampedY)
        end
    end)

    -- ===== 按钮事件 =====
    flyBtn.MouseButton1Click:Connect(function()
        if not flying then startFly() end
    end)
    stopBtn.MouseButton1Click:Connect(function()
        if flying then stopFly() end
    end)
    minBtn.MouseButton1Click:Connect(function()
        tweenPanel(not collapsed)
    end)
    closeBtn.MouseButton1Click:Connect(function()
        if flying then stopFly() end
        gui:Destroy()
    end)
end)

if not ok then
    warn("[辞飞行V3] UI 创建失败:", err)
    return
end

-- ===== 键盘快捷键 (PC端) =====
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F then
        toggleFly()
    elseif input.KeyCode == Enum.KeyCode.H then
        if panel and minBtn then
            tweenPanel(not collapsed)
        end
    elseif input.KeyCode == Enum.KeyCode.R then
        changeSpeed(flySpeed - FLY_SPEED_STEP)
    elseif input.KeyCode == Enum.KeyCode.T then
        changeSpeed(flySpeed + FLY_SPEED_STEP)
    end
end)

-- 角色重生时自动停止飞行
player.CharacterAdded:Connect(function()
    if flying then
        stopFly()
    end
end)
