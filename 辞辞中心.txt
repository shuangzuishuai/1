-- ===== 飞行逻辑 (V3 全游戏通用) =====

startFly = function()
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end

    for _, state in ipairs(humanoidStates) do
        hum:SetStateEnabled(state, false)
    end
    hum:ChangeState(Enum.HumanoidStateType.Swimming)

    local animate = char:FindFirstChild("Animate")
    if animate then animate.Disabled = true end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        track:AdjustSpeed(0)
    end

    hum.PlatformStand = true

    local isR6 = hum.RigType == Enum.HumanoidRigType.R6
    local bodyPart = isR6 and char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
    if not bodyPart then bodyPart = root end

    if bodyGyro then bodyGyro:Destroy() end
    bodyGyro = Instance.new("BodyGyro")
    bodyGyro.P = 9e4
    bodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    bodyGyro.CFrame = bodyPart.CFrame
    bodyGyro.Parent = bodyPart

    if bodyVelocity then bodyVelocity:Destroy() end
    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Velocity = Vector3.new(0, 0.1, 0)
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Parent = bodyPart

    tpwalking = true
    if hbConnection then hbConnection:Disconnect() end
    hbConnection = RunService.Heartbeat:Connect(function()
        if not tpwalking then return end
        local c = player.Character
        if not c then return end
        local h = c:FindFirstChildOfClass("Humanoid")
        if not h then return end
        local r = c:FindFirstChild("HumanoidRootPart")
        if not r then return end

        if h.MoveDirection.Magnitude > 0 then
            c:TranslateBy(h.MoveDirection * flySpeed)
        end

        local upDown = 0
        if upPressed then upDown = upDown + 1 end
        if downPressed then upDown = upDown - 1 end
        if not isTouch then
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then upDown = upDown + 1 end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then upDown = upDown - 1 end
        end
        if upDown ~= 0 then
            r.CFrame = r.CFrame * CFrame.new(0, upDown * flySpeed, 0)
        end

        if bodyGyro and bodyGyro.Parent then
            local cam = workspace.CurrentCamera
            if cam then bodyGyro.CFrame = cam.CFrame end
        end
    end)

    flying = true
    if upBtn then upBtn.Visible = true end
    if downBtn then downBtn.Visible = true end
    if flyBtn then flyBtn.BackgroundColor3 = Color3.fromRGB(80, 200, 100) end
    if stopBtn then stopBtn.BackgroundColor3 = Color3.fromRGB(120, 60, 65) end
    updateSpeedDisplay()
end

stopFly = function()
    local char = player.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            for _, state in ipairs(humanoidStates) do
                hum:SetStateEnabled(state, true)
            end
            hum:ChangeState(Enum.HumanoidStateType.RunningNoPhysics)
            hum.PlatformStand = false
        end
        local animate = char:FindFirstChild("Animate")
        if animate then animate.Disabled = false end
    end

    tpwalking = false
    if hbConnection then hbConnection:Disconnect() hbConnection = nil end
    if bodyGyro then bodyGyro:Destroy() bodyGyro = nil end
    if bodyVelocity then bodyVelocity:Destroy() bodyVelocity = nil end

    flying = false
    upPressed = false
    downPressed = false
    if upBtn then upBtn.Visible = false end
    if downBtn then downBtn.Visible = false end
    if flyBtn then flyBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 80) end
    if stopBtn then stopBtn.BackgroundColor3 = Color3.fromRGB(160, 70, 75) end
    updateSpeedDisplay()
end

toggleFly = function()
    if flying then stopFly() else startFly() end
end

updateSpeedDisplay = function()
    if not speedTextBox then return end
    if flying then
        speedTextBox.Text = tostring(flySpeed)
    else
        speedTextBox.Text = "0"
    end
end

changeSpeed = function(newSpeed)
    flySpeed = math.clamp(math.floor(newSpeed), FLY_SPEED_MIN, FLY_SPEED_MAX)
    updateSpeedDisplay()
end

-- ===== 移动速度逻辑 =====

applyWalkSpeed = function()
    local char = player.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then
            hum.WalkSpeed = walkSpeed
        end
    end
end

updateWalkSpeedDisplay = function()
    if walkSpeedTextBox then
        walkSpeedTextBox.Text = tostring(walkSpeed)
    end
end

changeWalkSpeed = function(newSpeed)
    walkSpeed = math.clamp(math.floor(newSpeed), WALK_SPEED_MIN, WALK_SPEED_MAX)
    applyWalkSpeed()
    updateWalkSpeedDisplay()
end

toggleWalkSpeedLock = function()
    walkSpeedLocked = not walkSpeedLocked
    if walkSpeedLocked then
        if walkSpeedLockConnection then walkSpeedLockConnection:Disconnect() end
        walkSpeedLockConnection = RunService.Heartbeat:Connect(function()
            applyWalkSpeed()
        end)
        if lockBtn then
            lockBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 80)
            lockBtn.Text = "已锁定"
        end
    else
        if walkSpeedLockConnection then
            walkSpeedLockConnection:Disconnect()
            walkSpeedLockConnection = nil
        end
        if lockBtn then
            lockBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 80)
            lockBtn.Text = "锁定速度"
        end
    end
end

removeESPForPlayer = function(p)
    local esp = espObjects[p]
    if esp then
        if esp.box then esp.box:Destroy() end
        if esp.name then esp.name:Destroy() end
        if esp.healthBg then esp.healthBg:Destroy() end
        if esp.healthText then esp.healthText:Destroy() end
        if esp.highlight then esp.highlight:Destroy() end
        if esp.billboard then esp.billboard:Destroy() end
        espObjects[p] = nil
    end
end

clearAllESP = function()
    for p, _ in pairs(espObjects) do
        removeESPForPlayer(p)
    end
end

local function hideESPObject(esp)
    if not esp then return end
    if esp.box then esp.box.Visible = false end
    if esp.name then esp.name.Visible = false end
    if esp.healthBg then esp.healthBg.Visible = false end
    if esp.healthText then esp.healthText.Visible = false end
    if esp.highlight then esp.highlight.Enabled = false end
    if esp.billboard then esp.billboard.Enabled = false end
end

toggleESP = function()
    espEnabled = not espEnabled
    if espEnabled then
        for _, p in ipairs(Players:GetPlayers()) do
            createESPForPlayer(p)
        end
        if espConnection then espConnection:Disconnect() end
        espConnection = RunService.RenderStepped:Connect(function()
            if not espEnabled then return end
            local cam = workspace.CurrentCamera
            if not cam then return end

            for p, esp in pairs(espObjects) do
                local char = p.Character
                if not char then
                    hideESPObject(esp)
                    continue
                end

                local hum = char:FindFirstChildOfClass("Humanoid")
                local root = char:FindFirstChild("HumanoidRootPart")
                local head = char:FindFirstChild("Head")

                if not hum or not root or not head then
                    hideESPObject(esp)
                    continue
                end

                -- 队友过滤
                if espIgnoreTeam and p.Team and player.Team and p.Team == player.Team then
                    hideESPObject(esp)
                    continue
                end

                local distance = (root.Position - cam.CFrame.Position).Magnitude
                local hp = math.clamp(hum.Health, 0, hum.MaxHealth)
                local maxHp = hum.MaxHealth
                if maxHp <= 0 then maxHp = 100 end
                local hpRatio = hp / maxHp

                if espHighlightMode then
                    -- 高亮模式：直接绑定 3D 角色，不用屏幕坐标，所以远近都不会下坠错位
                    if esp.box then esp.box.Visible = false end
                    if esp.name then esp.name.Visible = false end
                    if esp.healthBg then esp.healthBg.Visible = false end
                    if esp.healthText then esp.healthText.Visible = false end

                    if esp.highlight then
                        esp.highlight.Adornee = char
                        esp.highlight.Enabled = true
                        esp.highlight.FillColor = p.TeamColor and p.TeamColor.Color or Color3.fromRGB(0, 200, 255)
                    end

                    if esp.billboard then
                        esp.billboard.Adornee = head
                        esp.billboard.Enabled = espShowName or espShowHealth or espShowDistance
                        local bbText = espShowName and (p.DisplayName or p.Name) or ""
                        if espShowDistance then
                            if bbText ~= "" then bbText = bbText .. " " end
                            bbText = bbText .. "[" .. math.floor(distance) .. "m]"
                        end
                        if espShowHealth then
                            if bbText ~= "" then bbText = bbText .. " | " end
                            bbText = bbText .. "HP: " .. math.floor(hp) .. "/" .. math.floor(maxHp)
                        end
                        if esp.billboardText then
                            esp.billboardText.Text = bbText
                        end
                    end
                else
                    -- 旧 2D 方框模式：仅作为备用。默认关闭，避免远距离投影错位。
                    if esp.highlight then esp.highlight.Enabled = false end
                    if esp.billboard then esp.billboard.Enabled = false end

                    local rootPos, onScreen = cam:WorldToViewportPoint(root.Position)
                    if not onScreen then
                        hideESPObject(esp)
                        continue
                    end

                    local headPos2D = cam:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0))
                    local legPos2D = cam:WorldToViewportPoint(root.Position - Vector3.new(0, 3, 0))
                    local boxY = math.min(headPos2D.Y, legPos2D.Y)
                    local boxHeight = math.clamp(math.abs(legPos2D.Y - headPos2D.Y), 30, 500)
                    local boxWidth = boxHeight * 0.55
                    local boxX = headPos2D.X - boxWidth / 2

                    esp.box.Position = UDim2.new(0, boxX, 0, boxY)
                    esp.box.Size = UDim2.new(0, boxWidth, 0, boxHeight)
                    esp.box.Visible = espShowBox

                    local nameText = p.DisplayName or p.Name
                    if espShowDistance then
                        nameText = nameText .. " [" .. math.floor(distance) .. "m]"
                    end
                    esp.name.Text = nameText
                    esp.name.Position = UDim2.new(0, boxX - 10, 0, boxY - 16)
                    esp.name.Size = UDim2.new(0, boxWidth + 20, 0, 14)
                    esp.name.Visible = espShowName

                    esp.healthBg.Position = UDim2.new(0, boxX - 6, 0, boxY)
                    esp.healthBg.Size = UDim2.new(0, 3, 0, boxHeight)
                    esp.healthBg.Visible = espShowHealth

                    esp.healthFill.Size = UDim2.new(1, 0, hpRatio, 0)
                    esp.healthFill.Position = UDim2.new(0, 0, 1 - hpRatio, 0)
                    local r = math.floor(50 + 200 * (1 - hpRatio))
                    local g = math.floor(50 + 200 * hpRatio)
                    esp.healthFill.BackgroundColor3 = Color3.fromRGB(r, g, 0)

                    esp.healthText.Text = math.floor(hp) .. "/" .. math.floor(maxHp)
                    esp.healthText.Position = UDim2.new(0, boxX, 0, boxY + boxHeight + 2)
                    esp.healthText.Size = UDim2.new(0, boxWidth, 0, 12)
                    esp.healthText.Visible = espShowHealth
                end
            end
        end)

        if espToggleBtn then
            espToggleBtn.BackgroundColor3 = Color3.fromRGB(50, 160, 80)
            espToggleBtn.Text = "关闭透视"
        end
        if espHintLabel then
            espHintLabel.Text = "透视已开启 | 高亮绑定角色，远近不偏移"
        end
    else
        if espConnection then
            espConnection:Disconnect()
            espConnection = nil
        end
        for _, esp in pairs(espObjects) do
            hideESPObject(esp)
        end

        if espToggleBtn then
            espToggleBtn.BackgroundColor3 = Color3.fromRGB(50, 50, 70)
            espToggleBtn.Text = "开启透视"
        end
        if espHintLabel then
            espHintLabel.Text = "默认高亮跟随模式，远距离不下坠"
        end
    end
end

setESPOption = function(option)
    if option == "name" then
        espShowName = not espShowName
        return espShowName
    elseif option == "health" then
        espShowHealth = not espShowHealth
        return espShowHealth
    elseif option == "box" then
        espShowBox = not espShowBox
        return espShowBox
    elseif option == "distance" then
        espShowDistance = not espShowDistance
        return espShowDistance
    elseif option == "team" then
        espIgnoreTeam = not espIgnoreTeam
        return espIgnoreTeam
    elseif option == "highlight" then
        espHighlightMode = not espHighlightMode
        return espHighlightMode
    end
    return false
end

-- ===== UI 辅助 =====


local function fadeElement(el, toTransparent, duration)
    if not el then return end
    local target = toTransparent and 1 or 0
    local props = {}
    if el:IsA("TextLabel") or el:IsA("TextButton") or el:IsA("TextBox") then
        table.insert(props, { prop = "BackgroundTransparency", target = target })
        table.insert(props, { prop = "TextTransparency", target = toTransparent and 1 or 0 })
    elseif el:IsA("Frame") then
        table.insert(props, { prop = "BackgroundTransparency", target = target })
    end
    for _, p in ipairs(props) do
        TweenService:Create(el, TweenInfo.new(duration), { [p.prop] = p.target }):Play()
    end
end


-- 收起/展开面板（简化：只控制共享元素和页面容器的可见性）
tweenPanel = function(toCollapsed)
    if not panel or not minBtn then return end
    collapsed = toCollapsed
    local targetSize = collapsed and panelMiniSize or panelFullSize
    local tweenInfo = TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
    TweenService:Create(panel, tweenInfo, { Size = targetSize }):Play()

    if collapsed then
        -- 收起：淡出标签栏和所有页面
        if dividerLine then fadeElement(dividerLine, true, 0.2) end
        if flightTab then fadeElement(flightTab, true, 0.2) end
        if walkTab then fadeElement(walkTab, true, 0.2) end
        if espTab then fadeElement(espTab, true, 0.2) end
        if flightPage then flightPage.Visible = false end
        if walkPage then walkPage.Visible = false end
        if espPage then espPage.Visible = false end
        task.wait(0.2)
        if dividerLine then dividerLine.Visible = false end
        if flightTab then flightTab.Visible = false end
        if walkTab then walkTab.Visible = false end
        if espTab then espTab.Visible = false end
        minBtn.Text = "+"
    else
        -- 展开：显示标签栏 + 当前页
        if dividerLine then dividerLine.Visible = true end
        if flightTab then flightTab.Visible = true end
        if walkTab then walkTab.Visible = true end
        if espTab then espTab.Visible = true end
        -- 只显示当前页
        if flightPage then flightPage.Visible = (currentTab == "flight") end
        if walkPage then walkPage.Visible = (currentTab == "walk") end
        if espPage then espPage.Visible = (currentTab == "esp") end
        -- 淡入
        if dividerLine then fadeElement(dividerLine, false, 0.3) end
        if flightTab then fadeElement(flightTab, false, 0.3) end
        if walkTab then fadeElement(walkTab, false, 0.3) end
        if espTab then fadeElement(espTab, false, 0.3) end
        minBtn.Text = "—"
    end
end

-- 切换标签
switchTab = function(tabName)
    if tabName == currentTab then return end
    currentTab = tabName

    -- 隐藏所有页
    if flightPage then flightPage.Visible = false end
    if walkPage then walkPage.Visible = false end
    if espPage then espPage.Visible = false end

    -- 标签样式
    local function setTabStyle(tab, active)
        if not tab then return end
        if active then
            tab.BackgroundColor3 = Color3.fromRGB(50, 50, 70)
            tab.TextColor3 = Color3.fromRGB(255, 255, 255)
        else
            tab.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
            tab.TextColor3 = Color3.fromRGB(140, 140, 160)
        end
    end

    setTabStyle(flightTab, tabName == "flight")
    setTabStyle(walkTab, tabName == "walk")
    setTabStyle(espTab, tabName == "esp")

    -- 显示当前页
    if tabName == "flight" and flightPage then
        flightPage.Visible = true

-- ===== 键盘快捷键 (PC) =====
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if input.KeyCode == Enum.KeyCode.F then
        toggleFly()
    elseif input.KeyCode == Enum.KeyCode.H then
        if panel and minBtn then
            tweenPanel(not collapsed)
        end
    elseif input.KeyCode == Enum.KeyCode.R then
        changeSpeed(flySpeed - FLY_SPEED_STEP)
    elseif input.KeyCode == Enum.KeyCode.T then
        changeSpeed(flySpeed + FLY_SPEED_STEP)
    end
end)
